// ============================================================================
// ATENEA — LÓGICA DE DISTRIBUCIÓN AUTOMÁTICA FARMACIAS DEL AHORRO (FA)
// PROTOTIPO 1: "PLUS MEDIA CALLE NSE ALTO" — 29 pasos
// Traducción completa y literal del documento maestro de 29 pasos a C# compilable.
// ENTRADA ÚNICA: 4 puntos (P1, P2, P3, P4) en metros, en sentido
// P1(frente-izq) -> P2(frente-der) -> P3(atrás-der) -> P4(atrás-izq).
// Todo lo demás (Q2, Q3, C1-C7, B1, B2, D1, D2, E1-E3, F1-F6, G1-G7, J1, Cj1)
// se deriva matemáticamente de estos 4 puntos, recalculado desde cero.
// Uso: FA_GenerarSucursal.Generar(document, xP1,yP1, xP2,yP2, xP3,yP3, xP4,yP4);
//
// CORRECCIONES v1.9.1 (validadas en Revit 2025 contra caso real con muros
// inclinados, antes de aplicar a producción):
//   - PASO 21 (SOM): formula AnS corregida (signo de rotacion invertido respecto
//     a la inclinacion real del muro P2-P3).
//   - PASO 17 (Mobiliario tras mostrador): las familias se insertaban sin rotar;
//     ahora rotan con el angulo real de dirB1B2 (misma referencia que su posicion).
//   - PASO 28 (Cajon ADA): posicion ahora usa perpP1P2 (perpendicular real, no
//     dirP1P4) y rotacion final corregida a Angulo(P1->P2)+180° tras el espejo.
// ============================================================================

using System;
using System.Collections.Generic;
using Autodesk.Revit.DB;
using Autodesk.Revit.DB.Structure;

public static class FA_GenerarSucursal
{
    const double M = 3.280839895013123; // metros a pies

    // ---------- Utilidades geométricas ----------

    static XYZ Pt(double x, double y) { return new XYZ(x * M, y * M, 0); }

    static XYZ Interseccion(XYZ p1, XYZ d1, XYZ p2, XYZ d2)
    {
        double denom = d1.X * d2.Y - d1.Y * d2.X;
        double t = ((p2.X - p1.X) * d2.Y - (p2.Y - p1.Y) * d2.X) / denom;
        return p1 + d1 * t;
    }

    static XYZ Perp(XYZ dir) { return new XYZ(-dir.Y, dir.X, 0); }

    static double AnguloDe(XYZ dir) { return Math.Atan2(dir.Y, dir.X); }

    static double AnguloAB(XYZ a, XYZ b) { return Math.Atan2(b.Y - a.Y, b.X - a.X); }

    static FamilySymbol BuscarSimbolo(Document doc, string familia, string tipo)
    {
        FilteredElementCollector col = new FilteredElementCollector(doc).OfClass(typeof(FamilySymbol));
        foreach (Element e in col)
        {
            FamilySymbol fs = e as FamilySymbol;
            if (fs == null || fs.Family == null) continue;
            if (fs.Family.Name == familia && fs.Name == tipo) return fs;
        }
        return null;
    }

    static void Activar(Document doc, FamilySymbol sym)
    {
        if (sym != null && !sym.IsActive) sym.Activate();
    }

    static WallType BuscarTipoMuro(Document doc, string nombre)
    {
        FilteredElementCollector col = new FilteredElementCollector(doc).OfClass(typeof(WallType));
        foreach (Element e in col)
        {
            if (e.Name == nombre) return e as WallType;
        }
        return null;
    }

    static Level NivelBase(Document doc)
    {
        FilteredElementCollector col = new FilteredElementCollector(doc).OfClass(typeof(Level));
        Level nivel = null;
        double minElev = double.MaxValue;
        foreach (Element e in col)
        {
            Level lvl = e as Level;
            if (lvl != null && lvl.Elevation < minElev) { minElev = lvl.Elevation; nivel = lvl; }
        }
        return nivel;
    }

    static FamilyInstance Colocar(Document doc, XYZ centro, FamilySymbol sym, double anguloRad)
    {
        FamilyInstance fi = doc.Create.NewFamilyInstance(centro, sym, StructuralType.NonStructural);
        if (Math.Abs(anguloRad) > 1e-9)
            ElementTransformUtils.RotateElement(doc, fi.Id, Line.CreateBound(centro, centro + XYZ.BasisZ), anguloRad);
        return fi;
    }

    static int ColocarPatronG(Document doc, double[,] patron, XYZ puntoBase,
        XYZ perpP1P2, double anguloP1P2, FamilySymbol symCabecera, FamilySymbol symGondola)
    {
        int colocados = 0;
        int filas = patron.GetLength(0);
        for (int i = 0; i < filas; i++)
        {
            double dist = patron[i, 0];
            bool cab = patron[i, 1] > 0.5;
            double rot = patron[i, 2];
            XYZ centro = puntoBase + perpP1P2 * (dist * M);
            FamilySymbol sym = cab ? symCabecera : symGondola;
            double angulo = anguloP1P2 + (rot * Math.PI / 180.0);
            Colocar(doc, centro, sym, angulo);
            colocados++;
        }
        return colocados;
    }

    // ---------- Función principal ----------

    public static string Generar(
        Document document,
        double xP1, double yP1,
        double xP2, double yP2,
        double xP3, double yP3,
        double xP4, double yP4)
    {
        XYZ P1 = Pt(xP1, yP1);
        XYZ P2 = Pt(xP2, yP2);
        XYZ P3 = Pt(xP3, yP3);
        XYZ P4 = Pt(xP4, yP4);

        // ---------- VALIDACIÓN PREVIA ----------
        double Sa1chk = P3.DistanceTo(P4) / M;
        if (Sa1chk < 21.35 || Sa1chk > 23.75)
        {
            return "LAS MEDIDAS INGRESADAS NO CUMPLEN CON LOS ESTÁNDARES DE ESTE MODELO. " +
                   "Distancia P3-P4 ingresada: " + Sa1chk.ToString("F3") + " metros. " +
                   "Rango permitido: mínimo 21.35 metros, máximo 23.75 metros.";
        }

        int totalColocados = 0;
        int totalMuros = 0;

        using (SubTransaction t = new SubTransaction(document))
        {
            t.Start();

            // ===== PASO 1 =====
            XYZ dirP1P2 = (P2 - P1).Normalize();
            XYZ dirP1P4 = (P4 - P1).Normalize();

            // ===== PASO 2 =====
            XYZ dirP2P3 = (P3 - P2).Normalize();
            XYZ perpP2P3 = Perp(dirP2P3);
            if (perpP2P3.DotProduct(P1 - P2) < 0) perpP2P3 = perpP2P3.Negate();
            XYZ puntoOffsetQ = P2 + perpP2P3 * (2.95 * M);
            XYZ Q2 = Interseccion(puntoOffsetQ, dirP2P3, P1, dirP1P2);
            XYZ dirP3P4 = (P4 - P3).Normalize();
            XYZ Q3 = Interseccion(puntoOffsetQ, dirP2P3, P3, dirP3P4);
            XYZ dirQ2Q3 = (Q3 - Q2).Normalize();

            // ===== PASO 3 =====
            XYZ C1 = (P1 + Q2) / 2.0;

            // ===== PASO 4 =====
            XYZ dirP1Q2 = (Q2 - P1).Normalize();
            XYZ perpB = Perp(dirP1Q2);
            if (perpB.DotProduct(P4 - P1) < 0) perpB = perpB.Negate();
            XYZ puntoParalelaB = P1 + perpB * (7.96 * M);
            XYZ B1 = Interseccion(puntoParalelaB, dirP1Q2, P1, dirP1P4);
            XYZ B2 = Interseccion(puntoParalelaB, dirP1Q2, Q2, dirQ2Q3);
            XYZ dirB1B2 = (B2 - B1).Normalize();

            // ===== PASO 5 =====
            XYZ C2 = C1 - dirP1P2 * (1.48 * M);
            XYZ C3 = C1 + dirP1P2 * (1.48 * M);

            // ===== PASO 6 =====
            XYZ C4 = P1 + dirP1P2 * (2.33 * M);
            XYZ C5 = Q2 - dirP1P2 * (0.82 * M);

            // ===== PASO 7 =====
            XYZ C6 = (C4 + C2) / 2.0;
            XYZ C7 = (C3 + C5) / 2.0;

            // ===== PASO 9 (Muretes exteriores) =====
            WallType wtBloque = BuscarTipoMuro(document, "MU-BLOCK DE CONCRETO-EXTERIOR FARMACIA - 15CM");
            Level nivel = NivelBase(document);
            double altura = 3.0 * M;
            if (wtBloque == null || nivel == null)
            {
                t.RollBack();
                return "ERROR: no se encontró el tipo de muro MU-BLOCK DE CONCRETO-EXTERIOR FARMACIA - 15CM o ningún nivel.";
            }

            Wall.Create(document, Line.CreateBound(C5, Q2), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(Q2, P2), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(P2, P3), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(P3, P4), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(P4, P1), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(P1, C4), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(Q2, Q3), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;

            double mitadMurete = 0.18 * M;
            Wall.Create(document, Line.CreateBound(C6 - dirP1P2 * mitadMurete, C6 + dirP1P2 * mitadMurete), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(C7 - dirP1P2 * mitadMurete, C7 + dirP1P2 * mitadMurete), wtBloque.Id, nivel.Id, altura, 0, false, false); totalMuros++;

            // ===== PASO 10 (Muros interiores + fachada) =====
            WallType wtNovaceramic = BuscarTipoMuro(document, "MU-NOVACERAMIC INTERIOR-11CM");
            WallType wtCortina = BuscarTipoMuro(document, "MC-CORTINA DE CRISTAL CON MARCOS DE ALUMINIO");
            WallType wtDurock = BuscarTipoMuro(document, "MU-DUROCK 2 HOJAS INTERIOR-9.20CM");
            if (wtNovaceramic == null || wtCortina == null || wtDurock == null)
            {
                t.RollBack();
                return "ERROR: no se encontró uno de los tipos de muro del Paso 10 (Novaceramic, Cortina o Durock).";
            }

            XYZ par = dirP1P2;
            XYZ perpPar = Perp(par);
            double e041 = 0.41 * M, e032 = 0.32 * M, e082 = 0.82 * M;

            XYZ C3b = C3 + par * e032;
            XYZ c3A = C3b + perpPar * e041;
            XYZ c3B = c3A - par * e032;
            XYZ c3C = c3B - perpPar * e082;
            XYZ c3D = c3C + par * e032;
            Wall.Create(document, Line.CreateBound(C3b, c3A), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c3A, c3B), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c3B, c3C), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c3C, c3D), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c3D, C3b), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;

            XYZ c2A = C2 + perpPar * e041;
            XYZ c2B = c2A - par * e032;
            XYZ c2C = c2B - perpPar * e082;
            XYZ c2D = c2C + par * e032;
            Wall.Create(document, Line.CreateBound(C2, c2A), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c2A, c2B), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c2B, c2C), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c2C, c2D), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(c2D, C2), wtNovaceramic.Id, nivel.Id, altura, 0, false, false); totalMuros++;

            // Muros cortina (fachada + puerta)
            Wall.Create(document, Line.CreateBound(C4, C6 - par * (0.18 * M)), wtCortina.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(C6 + par * (0.18 * M), C2 - par * (0.38 * M)), wtCortina.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(C3 + par * (0.38 * M), C7 - par * (0.18 * M)), wtCortina.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(C7 + par * (0.18 * M), C5), wtCortina.Id, nivel.Id, altura, 0, false, false); totalMuros++;
            Wall.Create(document, Line.CreateBound(C1, C3), wtCortina.Id, nivel.Id, altura, 0, false, false); totalMuros++;

            // Muros Durock (division piso de tienda / trastienda) - requieren F4, F6 (Paso 15) y J1 (Paso 27), calculados mas abajo por eso se insertan al final del bloque de calculo de F. Ver mas abajo tras Paso 15/27.

            // ===== PASO 11 =====
            XYZ D1 = P1 + dirP1P4 * (0.81 * M);
            XYZ D2 = B1 - dirP1P4 * (0.55 * M);

            // ===== PASO 12 (Góndolas Derma Izq.) =====
            FamilySymbol symDerma = BuscarSimbolo(document, "FDA - GONDOLA DERMA", "0.30L X 0.46A X 2.43H");
            if (symDerma == null) { t.RollBack(); return "ERROR: no se encontró FDA - GONDOLA DERMA / 0.30L X 0.46A X 2.43H."; }
            Activar(document, symDerma);
            {
                double totalLen = D1.DistanceTo(D2) / M;
                double ancho = 0.34;
                int n = (int)Math.Floor(totalLen / ancho + 1e-6);
                double sobrante = totalLen - n * ancho;
                double start = sobrante / 2.0 + ancho / 2.0;
                XYZ dirRow = (D2 - D1).Normalize();
                XYZ perp = new XYZ(dirRow.Y, -dirRow.X, 0);
                double angulo = AnguloDe(perp) + (Math.PI / 2.0);
                for (int i = 0; i < n; i++)
                {
                    double dist = (start + i * ancho) * M;
                    XYZ centro = D1 + dirRow * dist + perp * (0.40 * M);
                    Colocar(document, centro, symDerma, angulo);
                    totalColocados++;
                }
            }

            // ===== PASO 13 =====
            XYZ E2 = Q2 + dirQ2Q3 * (2.97 * M);
            XYZ E3 = B2 - dirQ2Q3 * (0.55 * M);

            // ===== PASO 14 (Góndolas de Pared) =====
            FamilySymbol symPared = BuscarSimbolo(document, "FDA - GONDOLA DE PARED", "GONDOLA DE PARED 0.60L X 0.46A X 2.43H");
            if (symPared == null) { t.RollBack(); return "ERROR: no se encontró FDA - GONDOLA DE PARED / GONDOLA DE PARED 0.60L X 0.46A X 2.43H."; }
            Activar(document, symPared);
            {
                double totalLen = E2.DistanceTo(E3) / M;
                double ancho = 0.638;
                int n = (int)Math.Floor(totalLen / ancho + 1e-6);
                double sobrante = totalLen - n * ancho;
                double start = sobrante / 2.0 + ancho / 2.0;
                XYZ dirRow = (E3 - E2).Normalize();
                XYZ perp = new XYZ(-dirRow.Y, dirRow.X, 0);
                double angulo = AnguloDe(perp) + (Math.PI / 2.0);
                for (int i = 0; i < n; i++)
                {
                    double dist = (start + i * ancho) * M;
                    XYZ centro = E2 + dirRow * dist + perp * (0.40 * M);
                    Colocar(document, centro, symPared, angulo);
                    totalColocados++;
                }
            }

            // ===== PASO 15 (F1-F6) =====
            double Sa1 = P3.DistanceTo(P4) / M;
            double Sa2 = B2.DistanceTo(B1) / M;
            double Di1 = 0.57, Di2 = 0.57, Di4 = 0.57, Di5 = 9.01, Di6 = 0.85;
            double sumaConstantes = Di1 + Di2 + Di4 + Di5 + Di6;
            double Di3, Di7;
            if (Sa1 >= 21.35 && Sa1 < 21.95) { Di7 = 4.04; Di3 = Sa2 - sumaConstantes - Di7; }
            else { Di3 = 3.39; Di7 = Sa2 - sumaConstantes - Di3; }

            XYZ F2 = B1 + dirB1B2 * (Di1 * M);
            XYZ F1 = B2 - dirB1B2 * (Di2 * M);
            XYZ F3 = F2 + dirB1B2 * (Di3 * M);
            XYZ F4 = F3 + dirB1B2 * (Di4 * M);
            XYZ F5 = F4 + dirB1B2 * (Di5 * M);
            XYZ F6 = F5 + dirB1B2 * (Di6 * M);

            // ===== PASO 16 (Góndolas Derma atrás + Góndola de Pared) =====
            {
                XYZ perp = new XYZ(dirB1B2.Y, -dirB1B2.X, 0);
                double angulo = AnguloDe(perp) + (Math.PI / 2.0);
                double offsetPerp = 0.33 * M;

                double anchoDerma = 0.34;
                int nDerma = (int)Math.Floor(Di3 / anchoDerma + 1e-6);
                double sobranteDerma = Di3 - nDerma * anchoDerma;
                double startDerma = sobranteDerma / 2.0 + anchoDerma / 2.0;
                for (int i = 0; i < nDerma; i++)
                {
                    double dist = (startDerma + i * anchoDerma) * M;
                    XYZ centro = F2 + dirB1B2 * dist + perp * offsetPerp;
                    Colocar(document, centro, symDerma, angulo);
                    totalColocados++;
                }

                double anchoPared = 0.638;
                int nPared = (int)Math.Floor(Di7 / anchoPared + 1e-6);
                double sobrantePared = Di7 - nPared * anchoPared;
                double startPared = sobrantePared / 2.0 + anchoPared / 2.0;
                for (int i = 0; i < nPared; i++)
                {
                    double dist = (startPared + i * anchoPared) * M;
                    XYZ centro = F6 + dirB1B2 * dist + perp * offsetPerp;
                    Colocar(document, centro, symPared, angulo);
                    totalColocados++;
                }
            }

            // ===== PASO 17 (Mobiliario tras mostrador) =====
            // CORREGIDO v1.9.1: las familias se insertaban SIN rotacion, lo que las dejaba
            // en "dientes de sierra" en cuanto F4-F5 no era perfectamente horizontal. Ahora
            // se rota cada una con el angulo real de dirB1B2 (misma referencia que su posicion).
            {
                FamilySymbol symAnaquel = BuscarSimbolo(document, "ANAQUEL TRANSMOSTRADOR (2)", "Anaquel Transmostrador_1.20");
                FamilySymbol symPasgl = BuscarSimbolo(document, "FDA - PASGL ACCESO TRASTIENDA", "FDA - PASGL ACCESO TRASTIENDA");
                FamilySymbol symPantalla = BuscarSimbolo(document, "FDA - ANAQUEL TRANSMOSTRADOR CON PANTALLA", "Anaquel Transmostrador");
                if (symAnaquel == null || symPasgl == null || symPantalla == null)
                { t.RollBack(); return "ERROR: falta una familia del Paso 17 (Anaquel Transmostrador, Pasgl o Anaquel con Pantalla)."; }
                Activar(document, symAnaquel); Activar(document, symPasgl); Activar(document, symPantalla);

                XYZ perp17 = new XYZ(dirB1B2.Y, -dirB1B2.X, 0); // misma perpendicular usada en Paso 16, coherente con B1-B2
                double anguloB1B2_17 = AnguloDe(dirB1B2);
                var items17 = new (double dist, double off, FamilySymbol sym)[]
                {
                    (0.6717, 0.2624, symAnaquel),
                    (1.29,   0.60,   symPasgl),
                    (2.76,   0.2624, symAnaquel),
                    (3.96,   0.2624, symAnaquel),
                    (5.1604, 0.2624, symAnaquel),
                    (6.3603, 0.2624, symAnaquel),
                    (7.5791, 0.31,   symPantalla),
                    (8.2175, 0.60,   symPasgl),
                };
                List<ElementId> anaquelesInsertados = new List<ElementId>();
                foreach (var it in items17)
                {
                    XYZ centro = F4 + dirB1B2 * (it.dist * M) + perp17 * (it.off * M);
                    FamilyInstance fi = document.Create.NewFamilyInstance(centro, it.sym, StructuralType.NonStructural);
                    if (Math.Abs(anguloB1B2_17) > 1e-9)
                        ElementTransformUtils.RotateElement(document, fi.Id, Line.CreateBound(centro, centro + XYZ.BasisZ), anguloB1B2_17);
                    if (it.sym.Id == symAnaquel.Id) anaquelesInsertados.Add(fi.Id);
                    totalColocados++;
                }
                document.Regenerate();
                foreach (ElementId id in anaquelesInsertados)
                {
                    FamilyInstance fi = document.GetElement(id) as FamilyInstance;
                    if (fi == null) continue;
                    Parameter p1 = fi.LookupParameter("Bandeja activa 01");
                    Parameter p2 = fi.LookupParameter("VISIBILIDAD BANDEJA TRASERA PLANTA");
                    if (p1 != null && !p1.IsReadOnly) p1.Set(1);
                    if (p2 != null && !p2.IsReadOnly) p2.Set(1);
                }
            }

            // ===== PASO 18 (Puntos G1-G7) =====
            double G6x, G7x; bool existeG7;
            if (Sa1 >= 21.35 && Sa1 < 23.55) { G6x = 5.50 + (Sa1 - 21.35); G7x = 0; existeG7 = false; }
            else { G6x = 5.40 + (Sa1 - 23.75); G7x = 7.34 + (Sa1 - 23.75); existeG7 = true; }

            XYZ G1 = P1 + dirP1P2 * (2.08 * M) + dirP1P4 * (2.09 * M);
            XYZ G2 = P1 + dirP1P2 * (3.88 * M) + dirP1P4 * (2.39 * M);
            XYZ G3 = C3 + dirP1P2 * (-1.84 * M) + dirP1P4 * (2.00 * M);
            XYZ G4 = C3 + dirP1P4 * (2.00 * M);
            XYZ G5 = C3 + dirP1P2 * (1.84 * M) + dirP1P4 * (2.00 * M);
            XYZ G6 = C3 + dirP1P2 * (G6x * M) + dirP1P4 * (2.95 * M);
            XYZ G7 = C3 + dirP1P2 * (G7x * M) + dirP1P4 * (2.95 * M);

            // ===== PASO 19 (Islas Góndolas) =====
            {
                double Pb1 = P1.DistanceTo(B1) / M;
                double Pb2 = Q2.DistanceTo(B2) / M;
                bool caso1_1 = (Pb1 >= 7.96 && Pb1 < 8.60);
                bool caso2_1 = (Pb1 >= 8.60 && Pb1 < 8.80);
                bool caso1_2 = (Pb2 >= 7.96 && Pb2 < 8.60);
                bool caso2_2 = (Pb2 >= 8.60 && Pb2 < 8.80);

                FamilySymbol symCabecera = BuscarSimbolo(document, "CS12 CABECERA GONDOLA", "FENGPI- Góndola de Pared Inicial FENNEC");
                FamilySymbol symGondola = BuscarSimbolo(document, "FDA - GONDOLA", "GMS7I.CHO/GMS7A.CHO_1.370m");
                if (symCabecera == null || symGondola == null)
                { t.RollBack(); return "ERROR: falta CS12 CABECERA GONDOLA o FDA - GONDOLA del Paso 19."; }
                Activar(document, symCabecera); Activar(document, symGondola);

                double anguloP1P2 = AnguloDe(dirP1P2);
                XYZ perpP1P2 = Perp(dirP1P2);

                // Cada fila es {distancia, 1=cabecera/0=gondola, rotacion_relativa}
